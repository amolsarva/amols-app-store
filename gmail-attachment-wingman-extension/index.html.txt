  <body>
    <h1>Gmail Attachment Extractor</h1>
    <div class="note">
      Enter a Gmail search query. The script will copy matching attachments to Drive,
      label the original emails, and send bundle emails to you with the files attached.
      It cannot actually remove attachments from old Gmail messages.
    </div>
    <form id="form">
      <label for="query">Gmail search query</label>
      <textarea id="query" name="query" placeholder='Example: from:someone@example.com has:attachment newer_than:30d'></textarea>
      <label for="recipient">Send bundle emails to</label>
      <input id="recipient" name="recipient" type="email" placeholder="your@email.com">
      <button id="runButton" type="submit">Extract attachments</button>
    </form>
    <div class="examples">
      Useful examples:<br>
      has:attachment newer_than:7d<br>
      from:lawyer@example.com has:attachment<br>
      subject:NSCC has:attachment<br>
      filename:pdf NSCC<br>
      newer_than:30d (invoice OR receipt) has:attachment
    </div>
    <div id="status"></div>
    <script>
      const form = document.getElementById('form');
      const status = document.getElementById('status');
      const runButton = document.getElementById('runButton');
      form.addEventListener('submit', function(e) {
        e.preventDefault();
        const query = document.getElementById('query').value.trim();
        const recipient = document.getElementById('recipient').value.trim();
        if (!query) {
          status.innerHTML = '<div class="error">Please enter a Gmail search query.</div>';
          return;
        }
        runButton.disabled = true;
        runButton.textContent = 'Working...';
        status.innerHTML = '<div class="result">Scanning Gmail and extracting attachments. Keep this tab open.</div>';
        google.script.run
          .withSuccessHandler(function(result) {
            runButton.disabled = false;
            runButton.textContent = 'Extract attachments';
            status.innerHTML = `
              <div class="result">
                <strong>Done.</strong><br><br>
                Search query: <code>${escapeHtml(result.query)}</code><br>
                Sent bundle emails to: ${escapeHtml(result.recipient)}<br>
                Label applied: ${escapeHtml(result.label)}<br><br>
                Matching threads: ${result.totalThreads}<br>
                Messages scanned: ${result.totalMessages}<br>
                Attachments found: ${result.totalAttachments}<br>
                Saved to Drive: ${result.totalSavedToDrive}<br>
                Attached to sent emails: ${result.totalAttachedToEmails}<br>
                Bundle emails sent: ${result.emailBundleCount}<br><br>
                <a href="${result.folderUrl}" target="_blank">Open Drive folder</a><br>
                <a href="${result.manifestUrl}" target="_blank">Open manifest CSV</a>
              </div>`;
          })
          .withFailureHandler(function(error) {
            runButton.disabled = false;
            runButton.textContent = 'Extract attachments';
            status.innerHTML = `<div class="error"><strong>Error:</strong><br>${escapeHtml(error.message || String(error))}</div>`;
          })
          .runAttachmentExtractor({ query, recipient });
      });
      function escapeHtml(value) {
        return String(value || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#039;');
      }
    </script>
